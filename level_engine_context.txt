133:     if not lowered:
134:         return ""

198: def evaluate_level_requirement(
199:     level: int,
200:     package_name: Any,
201:     active_direct_customers: Any,
202:     purchased_courses: Optional[Iterable[Any]],
203: ) -> dict:
204:     if level not in LEVEL_REQUIREMENTS:
205:         raise ValueError(f"Unsupported level: {level}")
206: 
207:     requirement = LEVEL_REQUIREMENTS[level]
208:     package = normalize_package_name(package_name)
209:     active_customers = _safe_int(active_direct_customers)
210:     courses = normalize_courses(purchased_courses)
211: 
212:     package_ok = package in requirement.allowed_packages
213:     customers_ok = active_customers >= requirement.min_active_direct_customers
214:     courses_ok = all(course in courses for course in requirement.required_courses)
215: 
216:     missing_courses = [
217:         course for course in requirement.required_courses
218:         if course not in courses
219:     ]
220: 
221:     missing_active_direct_customers = max(
222:         requirement.min_active_direct_customers - active_customers,
223:         0,
224:     )
225: 
226:     return {
227:         "level": requirement.level,
228:         "name": requirement.name,
229:         "commission_rate": requirement.commission_rate,
230:         "allowed_packages": list(requirement.allowed_packages),
231:         "current_package": package,
232:         "package_ok": package_ok,
233:         "min_active_direct_customers": requirement.min_active_direct_customers,
234:         "active_direct_customers": active_customers,
235:         "customers_ok": customers_ok,
236:         "missing_active_direct_customers": missing_active_direct_customers,
237:         "required_courses": list(requirement.required_courses),
238:         "purchased_courses": sorted(courses),
239:         "courses_ok": courses_ok,
240:         "missing_courses": missing_courses,
241:         "qualified": bool(package_ok and customers_ok and courses_ok),
242:     }

245: def calculate_partner_level_progress(
246:     package_name: Any,
247:     active_direct_customers: Any = 0,
248:     purchased_courses: Optional[Iterable[Any]] = None,
249:     subscription_status: Any = "active",
250: ) -> dict:
251:     package = normalize_package_name(package_name)
252:     status = normalize_subscription_status(subscription_status)
253:     subscription_active = is_subscription_active(status)
254:     active_customers = _safe_int(active_direct_customers)
255:     courses = normalize_courses(purchased_courses)
256: 
257:     achieved_level = 0
258:     level_details = []
259: 
260:     for level in range(1, 6):
261:         detail = evaluate_level_requirement(
262:             level=level,
263:             package_name=package,
264:             active_direct_customers=active_customers,
265:             purchased_courses=courses,
266:         )
267: 
268:         if detail["qualified"] and achieved_level == level - 1:
269:             achieved_level = level
270: 
271:         level_details.append(detail)
272: 
273:     next_level = achieved_level + 1 if achieved_level < 5 else None
274:     missing_requirements = []
275: 
276:     if next_level:
277:         next_detail = level_details[next_level - 1]
278: 
279:         if not next_detail["package_ok"]:
280:             missing_requirements.append({
281:                 "type": "package",
282:                 "current": package,
283:                 "required": next_detail["allowed_packages"],
284:             })
285: 
286:         if not next_detail["customers_ok"]:
287:             missing_requirements.append({
288:                 "type": "active_direct_customers",
289:                 "current": active_customers,
290:                 "required": next_detail["min_active_direct_customers"],
291:                 "missing": next_detail["missing_active_direct_customers"],
292:             })
293: 
294:         if not next_detail["courses_ok"]:
295:             missing_requirements.append({
296:                 "type": "courses",
297:                 "current": sorted(courses),
298:                 "required": next_detail["required_courses"],
299:                 "missing": next_detail["missing_courses"],
300:             })
301: 
302:     return {
303:         "current_package": package,
304:         "subscription_status": status,
305:         "subscription_active": subscription_active,
306:         "active_direct_customers": active_customers,
307:         "purchased_courses": sorted(courses),
308:         "achieved_level": achieved_level,
309:         "current_level": achieved_level,
310:         "current_level_name": LEVEL_NAMES.get(achieved_level, "Unknown"),
311:         "commission_eligible": bool(subscription_active and achieved_level >= 1),
312:         "next_level": next_level,
313:         "next_level_name": LEVEL_NAMES.get(next_level) if next_level else None,
314:         "missing_requirements": missing_requirements,
315:         "level_details": level_details,
316:     }

319: def is_partner_eligible_for_commission_depth(
320:     current_level: Any,
321:     commission_depth: Any,
322:     subscription_status: Any = "active",
323: ) -> bool:
324:     level = _safe_int(current_level)
325:     depth = _safe_int(commission_depth)
326: 
327:     if depth < 1 or depth > 5:
328:         return False
329: 
330:     if not is_subscription_active(subscription_status):
331:         return False
332: 
333:     return level >= depth

341: def summarize_next_level_requirement(progress: dict) -> str:
342:     next_level = progress.get("next_level")
343: 
344:     if not next_level:
345:         return "You already reached the highest level."
346: 
347:     missing = progress.get("missing_requirements") or []
348: 
349:     if not missing:
350:         return f"You are eligible for Level {next_level}."
351: 
352:     parts = []
353: 
354:     for item in missing:
355:         if item.get("type") == "package":
356:             parts.append("upgrade package")
357:         elif item.get("type") == "active_direct_customers":
358:             parts.append(f"{item.get('missing', 0)} more active direct customers")
359:         elif item.get("type") == "courses":
360:             parts.append("complete required course purchase")
361:         else:
362:             parts.append("missing requirement")
363: 
364:     return f"To reach Level {next_level}, you need: " + ", ".join(parts)
